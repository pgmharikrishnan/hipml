"""Fetch a handful of pages and report their titles.

Run with: python crawl.py https://example.com https://example.org
"""

import argparse
import re
import sys
import urllib.error
import urllib.request


DEFAULT_TIMEOUT = 10.0
TITLE_RE = re.compile(r"<title[^>]*>(.*?)</title>", re.IGNORECASE | re.DOTALL)


class FetchError(Exception):
    """Raised when a page cannot be fetched or decoded."""


def parse_args(argv):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("urls", nargs="+", help="pages to fetch")
    parser.add_argument(
        "--timeout",
        type=float,
        default=DEFAULT_TIMEOUT,
        help="socket timeout in seconds (default: %(default)s)",
    )
    parser.add_argument(
        "--user-agent",
        default="hipml-crawler/1.0",
        help="User-Agent header (default: %(default)s)",
    )
    return parser.parse_args(argv)


def fetch_page(url, timeout=DEFAULT_TIMEOUT, user_agent="hipml-crawler/1.0"):
    """Fetch one URL and return the decoded body."""
    request = urllib.request.Request(url, headers={"User-Agent": user_agent})
    try:
        with urllib.request.urlopen(request, timeout=timeout) as response:
            charset = response.headers.get_content_charset() or "utf-8"
            return response.read().decode(charset, errors="replace")
    except (urllib.error.URLError, TimeoutError, OSError) as exc:
        raise FetchError(f"{url}: {exc}") from exc


def extract_title(html):
    """Return the first <title> text, or an empty string."""
    match = TITLE_RE.search(html)
    if not match:
        return ""
    title = re.sub(r"\s+", " ", match.group(1)).strip()
    return title[:120]


def main(argv=None):
    args = parse_args(argv)
    failures = 0
    for url in args.urls:
        try:
            html = fetch_page(url, timeout=args.timeout, user_agent=args.user_agent)
        except FetchError as exc:
            print(f"FAIL {exc}", file=sys.stderr)
            failures += 1
            continue
        title = extract_title(html) or "(untitled)"
        print(f"OK   {title}  [{len(html)} bytes]  {url}")
    return 1 if failures else 0


if __name__ == "__main__":
    raise SystemExit(main())
